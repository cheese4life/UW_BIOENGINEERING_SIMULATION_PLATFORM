dir (directoryPath)
Returns a table containing a directory listing. If the directory path is invalid, an error is generated.

Parameters:

  • directoryPath string The path to the directory.

Returns:

    table A table (ie. a list of strings) of directory entries.

Usage:

    local log = require 'lua_logger'
local filesystem = require 'lua_filesystem'

-- Retrieve the contents of the current working directory.
local dirListing = filesystem.dir('.')
local maxFilesToDisplay = 5

log.info('Displaying the first ' .. maxFilesToDisplay .. ' of '
  .. #dirListing .. ' files.')

for i,filename in pairs(dirListing) do
  log.info(filename)
  maxFilesToDisplay = maxFilesToDisplay - 1
  if maxFilesToDisplay == 0 then
    break
  end
end
direxists (directoryPath)
Checks for the existence of a directory.

Parameters:

  • directoryPath string The path to the directory.

Returns:

    boolean Returns true if the directory exists, false if not.

Usage:

    local log = require 'lua_logger'
local filesystem = require 'lua_filesystem'

-- Check if a directory exists.
local directoryName = [[C:\Temp]]
local doesDirectoryExist = filesystem.dir(directoryName)

log.info('Directory ' .. directoryName .. ' exists: ' .. doesDirectoryExist .. '.')
filename (pathname)
Returns a string containing the filename from a given filepath.

Parameters:

  • pathname string A full pathname which includes directories and a filename.

Returns:

    string The filename part of a path, including the filename extension.

Usage:

    local filesystem = require 'lua_filesystem'

local pathname = [[C:\Program Files\CompanyName\Application\Instrument.cfg]]
local filename = filesystem.filename(pathname)
-- filename contains "Instrument.cfg"

resolvepathname (pathname)
Parses a pathname by replacing any template items with their runtime paths.

Parameters:

  • pathname string A full pathname which includes template items.

Returns:

    string A fully resolved pathname which has its template item replaced with a real pathname.

Usage:

    local filesystem = require 'lua_filesystem'

-- Path resolution is specific to how the application is set up.
-- For this example, assume the application sets up [SupportFolder] to
-- the path "C:\IMPApplication\SupportFolder"
local pathname = [[[SupportFolder]\Instrument.cfg]]
local resolvedPathname = filesystem.resolvepathname(pathname)
-- this returns "C:\IMPApplication\SupportFolder\Instrument.cfg"
